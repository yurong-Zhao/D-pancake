// ========== 加载全局数据 ==========

async function loadGlobalLists() {

    try {

        // 加载IP列表

        const ipSnapshot = await db.collection("global_ips").get();

        if (!ipSnapshot.empty) {

            globalIpList = [];

            ipSnapshot.forEach(doc => globalIpList.push(doc.data().name));

        } else {

            globalIpList = [];

            await db.collection("global_ips").doc("default").set({name:"默认IP"});

            globalIpList.push("默认IP");

        }

        // 加载成员列表

        const memberSnapshot = await db.collection("global_members").get();

        if (!memberSnapshot.empty) {

            globalMemberList = [];

            memberSnapshot.forEach(doc => globalMemberList.push(doc.data().name));

        } else {

            for (let m of globalMemberList) {

                await db.collection("global_members").doc(m).set({name:m});

            }

        }

        refreshIpSelect();

        refreshMemberCheck();

    } catch(e) {

        console.error("加载全局数据失败", e);

    }

}


async function saveNewIp(ip) {

    if (!ip || globalIpList.includes(ip)) return;

    globalIpList.push(ip);

    await db.collection("global_ips").doc(ip).set({name:ip});

    refreshIpSelect();

}


async function saveNewMember(name) {

    if (!name || globalMemberList.includes(name)) return;

    globalMemberList.push(name);

    await db.collection("global_members").doc(name).set({name:name});

    refreshMemberCheck();

}


function refreshIpSelect() {

    let sel = document.getElementById("existIpSel");

    sel.innerHTML = '<option value="">选择已有IP</option>';

    globalIpList.forEach(ip => {

        sel.innerHTML += `<option value="${ip}">${ip}</option>`;

    });

}


function refreshMemberCheck() {

    let box = document.getElementById("memberCheckBox");

    box.innerHTML = "";

    globalMemberList.forEach(name => {

        box.innerHTML += `<label class="flex items-center gap-1 border px-2 py-1 rounded cursor-pointer"><input type="checkbox" value="${name}" onchange="changeMemberSelect()"> ${name}</label>`;

    });

}

function autoFillAddressByIp() {

    let date = document.getElementById("planDateSel").value;

    let selIp = document.getElementById("existIpSel").value;

    if (!selIp) return;

    let sameIpInfo = planData.find(item => item.date === date && item.ip === selIp);

    if (!sameIpInfo) return;

    document.getElementById("provinceSel").value = sameIpInfo.province || "";

    changeProvince();

    setTimeout(() => {

        document.getElementById("citySel").value = sameIpInfo.city || "";

    }, 50);

    document.getElementById("addressInput").value = sameIpInfo.address || "";

}

                    </select>

                    <input type="text" class="mem-role border rounded p-1 w-full mt-1" data-name="${uname}" placeholder="自定义扮演角色">

    });

    wrap.innerHTML = html;

    let plan = planData.find(p => String(p.id) === String(planId));

    if (!plan) return;

    let member = plan.memberList.find(m => m.userName === memberName);

    if (!member) return;

    if (member.userStatus === "已准备齐全") {

        alert("该成员已准备齐全，无法催促");

        return;

    }

    member.urgeCount = (member.urgeCount || 0) + 1;

    if (member.urgeCount > 10) member.urgeCount = 10;

    await syncSinglePlanToFirestore(plan);

    renderPlanList();

    renderCalendar();

}


function getUrgeIcons(count) {

    if (!count || count === 0) return "无催促";

    let icons = "";

    for (let i = 0; i < Math.min(count, 10); i++) icons += "🔥";

    return icons;

}


async function changeMemberStatusInList(planId, memberName, newStatus) {

    let plan = planData.find(p => String(p.id) === String(planId));

    if (!plan) return;

    let member = plan.memberList.find(m => m.userName === memberName);

    if (!member) return;

    member.userStatus = newStatus;

    if (newStatus === "已准备齐全") {

        member.urgeCount = 0;

    }

    await syncSinglePlanToFirestore(plan);

    renderPlanList();

    renderCalendar();

}


async function syncSinglePlanToFirestore(plan) {

    const planRef = db.collection("plans").doc(String(plan.id));

    const { id, ...data } = plan;

    await planRef.set(data);

}

async function saveMemberEdit() {

    let r = document.getElementById("editRoleInput").value.trim();

    let idx = currentViewPlan.memberList.findIndex(x => x.userName === editTargetMemberName);

    if (idx > -1) {

            currentViewPlan.memberList[idx].userStatus = "无需准备状态";

        }

    let mainIdx = planData.findIndex(p => String(p.id) === String(currentViewPlan.id));

    if (mainIdx > -1) planData[mainIdx].memberList = currentViewPlan.memberList;

    await syncSinglePlanToFirestore(currentViewPlan);

    closeMemberEdit();

    renderPlanList();

    alert("修改完成");

}

async function saveGroupPlan() {

    let date = document.getElementById("planDateSel").value;

    let selIp = document.getElementById("existIpSel").value;

    let newIp = document.getElementById("newIpInput").value.trim();

    let ipName = selIp || newIp;

    if (!date || !ipName) { alert("请选择日期并填写IP"); return; }

    if (newIp) await saveNewIp(newIp);

    

    let type = document.getElementById("planTypeSel").value;

    let province = document.getElementById("provinceSel").value;

    let city = document.getElementById("citySel").value;

    let address = document.getElementById("addressInput").value.trim();

    

    let memberArr = [];

    document.querySelectorAll(".mem-role").forEach(ri => {

        let un = ri.dataset.name;

        let ro = ri.value.trim();

        if (!ro) ro = "待定";

        let st = document.querySelector(`.mem-status[data-name="${un}"]`)?.value || "未确定角色";

        memberArr.push({ userName: un, userRole: ro, userStatus: st, urgeCount: 0 });

    });

    if (memberArr.length === 0) { alert("请勾选成员"); return; }

    

    let newPlan = {

        id: Date.now(),

        date, ip: ipName, planType: type,

        province, city, address,

        memberList: memberArr

    };

    planData.push(newPlan);

    await db.collection("plans").doc(String(newPlan.id)).set(newPlan);

    closeAddModal();

    renderCalendar();

    renderPlanList();

    alert("保存成功");

}


// ========== 日历渲染 ==========

function renderCalendar() {

    let monthText = document.getElementById("curMonthText");

    if (monthText) monthText.innerText = `${showYear}年${showMonth+1}月`;

    let firstDay = new Date(showYear, showMonth, 1).getDay();

    let totalDay = new Date(showYear, showMonth+1, 0).getDate();

    let box = document.getElementById("calendarBox");

    if (!box) return;

    box.innerHTML = "";

    for (let i = 0; i < firstDay; i++) {

        box.innerHTML += `<div class="calendar-day bg-gray-50 rounded p-2"></div>`;

    }

    for (let d = 1; d <= totalDay; d++) {

        let dateStr = `${showYear}-${String(showMonth+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;

        let dayPlans = planData.filter(p => p.date === dateStr);

        let hasPlan = dayPlans.length > 0;

        let planIcons = "";

        if (hasPlan) {

            let allUrgeCount = 0;

            dayPlans.forEach(p => {

                if (p.memberList) {

                    p.memberList.forEach(m => { allUrgeCount += (m.urgeCount || 0); });

                }

            });

            planIcons = allUrgeCount > 0 ? `<span class="text-xs">🔥${allUrgeCount}</span>` : `<span class="tag-dot bg-blue-500 inline-block"></span>`;

        }

        box.innerHTML += `

        <div class="calendar-day bg-white border rounded p-2 cursor-pointer hover:border-blue-400" onclick="openDatePlanView('${dateStr}')">

            <div class="font-medium">${d}</div>

            ${hasPlan ? `<div class="flex gap-1 mt-1">${planIcons}</div>` : ""}

        </div>`;

    }

}


function openDatePlanView(date) {

    let arr = planData.filter(p => p.date === date);

    if (!arr.length) { alert("当日暂无计划"); return; }

    let map = {};

    arr.forEach(i => {

        let k = `${i.date}|${i.ip}|${i.planType}`;

        if (!map[k]) map[k] = { ...i, memberList: [] };

        map[k].memberList.push(...i.memberList);

    });

    openPlanDetail(Object.values(map)[0]);

}


function prevMonth() { showMonth--; if (showMonth < 0) { showMonth = 11; showYear--; } renderCalendar(); }

function nextMonth() { showMonth++; if (showMonth > 11) { showMonth = 0; showYear++; } renderCalendar(); }
